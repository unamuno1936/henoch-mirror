      🏗️ Arquitectura de Trabajo: Las 3 Capas


        Capa 1: El Registro Inmediato (La Bitácora Viva)

  *

    *Herramienta:*|worklogs|(Python)-63 <https://pypi.org/project/
    worklogs/#1>.

  *

    *Por qué:*Es una herramienta de línea de comandos que crea archivos
    Markdown locales con una estructura de carpetas por fecha (|.../
    YYYY/MM-month/DD-ddd/HHMM-Ha--nombre--tipo.md|)-63 <https://
    pypi.org/project/worklogs/#1>. Es la forma más rápida y de menor
    fricción de anotar una idea, un fallo o un avance, sin salir de la

        Capa 2: El Cerebro del Proyecto (Gestión y Auditoría)

  *

    *Herramienta:*|Clew|(Rust)-13 <https://docs.rs/crate/clew/0.1.0#1>.

  *

    *Por qué:*Es un rastreador de proyectos *git-nativo*, local y
    ligero, diseñado para ser usado por humanos y agentes de IA por
    igual-13 <https://docs.rs/crate/clew/0.1.0#1>. Sus conceptos de
    *"Incremento"*y *"Relé"*son clave:

      o

        *Incremento:*Una unidad de trabajo que, al completarse, deja el
        código en un estado estable y commiteable-13 <https://docs.rs/
        crate/clew/0.1.0#1>.

      o

        *Relé:*Captura el contexto entre sesiones de trabajo, incluyendo

        repositorio, y el "Relé" es tu punto de partida para la
        siguiente. Al ser git-nativo, el historial de decisiones vive
        junto al código.


        Capa 3: La Vidriera (Visibilidad Selectiva y Espejos)

  *

    *Herramienta:*|HEADBANG|(Node.js)-40 <https://www.npmjs.com/package/
    headbang-mcp#1>.

  *

    *Por qué:*Es un *enrutador de políticas para Git*. Te permite tomar
    un repositorio local y publicar /proyecciones/diferentes en

    defines-40 <https://www.npmjs.com/package/headbang-mcp#1>. Puedes

    🐍 Parte 3: Script Python Unificado

Aquí tienes un script que integra grabación, transcripción y
verificación. Guárdalo como |consulta_voz.py|:

python

Copy

Download

#!/usr/bin/env python3
"""
consulta_voz.py — Graba audio, transcribe con whisper.cpp y verifica límites.
Parte del pipeline de bitácora del Proyecto Enoch.
"""

import subprocess
import sys
import os
from datetime import datetime
from pathlib import Path

# ═══════════════════════════════════════════
# CONFIGURACIÓN
# ═══════════════════════════════════════════
WHISPER_BIN = os.path.expanduser("~/herramientas/whisper.cpp/build/bin/whisper-cli")
WHISPER_MODEL = os.path.expanduser("~/herramientas/whisper.cpp/models/ggml-small.bin")
DIR_BITACORA = Path(os.path.expanduser("~/proyecto/bitacora/audios"))
DURACION_MAX = 60       # segundos
LIMITE_PALABRAS = 120   # palabras

DIR_BITACORA.mkdir(parents=True, exist_ok=True)

def grabar_audio(segundos: int, nombre: str) -> Path:
    """Graba audio del micrófono usando arecord."""
    archivo = DIR_BITACORA / f"{nombre}.wav"
    print(f"🔴 Grabando {segundos}s... Habla ahora.")
    subprocess.run(
        ["arecord", "-f", "cd", "-t", "wav", "-d", str(segundos), str(archivo)],
        check=True
    )
    print(f"✅ Audio: {archivo}")
    return archivo

def transcribir(archivo_wav: Path, idioma: str = "es") -> str:
    """Transcribe con whisper.cpp y devuelve el texto."""
    archivo_txt = archivo_wav.with_suffix(".txt")
    cmd = [
        WHISPER_BIN,
        "-m", WHISPER_MODEL,
        "-f", str(archivo_wav),
        "-l", idioma,
        "-ml", "80",
        "-otxt",
        "-of", str(archivo_wav.with_suffix(""))
    ]
    print("🧠 Transcribiendo...")
    subprocess.run(cmd, check=True, capture_output=True)

    if not archivo_txt.exists():
        raise FileNotFoundError(f"No se generó: {archivo_txt}")

    texto = archivo_txt.read_text(encoding="utf-8").strip()
    return texto

def verificar_limite(texto: str, limite: int = LIMITE_PALABRAS) -> bool:
    """Verifica que el texto no exceda el límite de palabras."""
    palabras = len(texto.split())
    print(f"📊 Palabras: {palabras} / {limite}")
    if palabras > limite:
        print("⚠️  EXCEDE el límite.")
        print(f"   Sugerencia: divide en Contexto + Petición.")
        return False
    print("✅ Dentro del límite.")
    return True

def guardar_en_bitacora(texto: str, archivo_audio: Path):
    """Guarda la transcripción con metadatos en la bitácora."""
    fecha = datetime.now().strftime("%Y-%m-%d_%H%M")
    entrada = DIR_BITACORA / f"entrada_{fecha}.md"

    contenido = f"""# Entrada de voz — {fecha}

**Audio:** `{archivo_audio.name}`
**Palabras:** {len(texto.split())}

---

{texto}
"""
    entrada.write_text(contenido, encoding="utf-8")
    print(f"📝 Bitácora: {entrada}")
    return entrada

def main():
    nombre = f"consulta_{datetime.now().strftime('%Y%m%d_%H%M%S')}"

    try:
        audio = grabar_audio(DURACION_MAX, nombre)
        texto = transcribir(audio)
        print(f"\n{'─'*50}")
        print(texto)
        print(f"{'─'*50}\n")

        if verificar_limite(texto):
            guardar_en_bitacora(texto, audio)
        else:
            print("\n💡 Puedes editar el texto o dividirlo en dos entradas.")

    except FileNotFoundError as e:
        print(f"❌ Error: {e}", file=sys.stderr)
        sys.exit(1)
    except subprocess.CalledProcessError as e:
        print(f"❌ Error de ejecución: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()

    🧠 Parte 4: La Consulta Quirúrgica vs. La Consulta que Hace Volar al
    Modelo

Tu distinción es fundamental. No es lo mismo una *orden*que una
*provocación*. Un LLM no "vuela" con preguntas genéricas; vuela cuando
le das *restricciones precisas*y *contexto rico*.


      Consulta Quirúrgica (30-50 palabras)

*Características:*

  *

    Un solo objetivo.

  *

    Sin contexto innecesario.

  *

    Instrucción directa al final.

*Estructura:*|[Verbo] + [objeto] + [restricción]|

*Ejemplo:*

    "Verifica si esta cita de Dillmann coincide con Knibb. Si no
    coincide, señala la variante exacta. Cita: [texto]."

*Técnicas clave:*-50 <https://learn.microsoft.com/en-us/azure/foundry/
openai/concepts/prompt-engineering#2>

  *

    *Repite la instrucción al final*("...y si no coincide, señala la
    variante").

  *

    *Usa "priming"*: termina con "Respuesta:" o "```json" para forzar el
    formato.

  *

    *Sé específico sobre la salida*: "Devuelve solo un booleano y la
    variante".


      Consulta que Hace Volar al Modelo (100-150 palabras)

*Características:*

  *

    Contexto del proyecto (¿qué es Enoch? ¿qué estás haciendo?).

  *

    Una pregunta que invita a razonar, no solo a responder.

  *

    Restricciones que obligan a la profundidad.

*Estructura:*

text

Copy

Download

[Contexto: 2-3 frases sobre el proyecto y el estado actual]
[Pregunta abierta que invita al razonamiento]
[Restricciones: qué NO quieres, qué formato, qué nivel de detalle]

*Ejemplo:*

    "Estoy reconstruyendo la historia editorial de 1 Enoc. Acabo de
    documentar que Hoffmann (1833) tradujo los capítulos 1-55 del inglés
    de Laurence, no del Ge'ez. Quiero entender cómo esta mediación
    lingüística afectó la recepción alemana del texto. ¿Puedes
    identificar los pasajes donde la traducción de Laurence introduce
    ambigüedades que Hoffmann hereda? Restricción: no especules sin
    citar fuentes; si no tienes datos, dilo. Formato: lista de pasajes
    con referencia."

*Técnicas clave:*

  *

    *Persona*: "Actúa como un historiador de la transmisión textual"-50
    <https://learn.microsoft.com/en-us/azure/foundry/openai/concepts/
    prompt-engineering#2>.

  *

    *Pide reflexión*: "Antes de responder, enumera los supuestos que
    estás haciendo".

  *

    *Fricción productiva*: "Si detectas que mi premisa es errónea,
    corrígeme antes de responder".


      ¿Cómo "friccionar" un modelo sin caer en prompts fabricados?

La fricción no es hacer el prompt más largo. Es *obligar al modelo a
confrontar sus propias lagunas*. Tres tácticas:

 1.

    *Pídele que se contradiga a sí mismo*: "Genera dos respuestas: una
    que defienda X y otra que lo refute. Luego, señala cuál es más
    sólida y por qué."

 2.

    *Impón un formato que no pueda rellenar con generalidades*:
    "Devuelve una tabla con tres columnas: Afirmación | Evidencia |
    Nivel de confianza (Alto/Medio/Bajo). Si no tienes evidencia,
    escribe 'Sin fuente verificable'."

 3.

    *Usa el silencio como fricción*: "Si no tienes datos suficientes
    para responder con precisión, responde únicamente: |[SIN DATOS
    SUFICIENTES]|. No inventes."

Esto se alinea con tu principio operativo: *no inventar*. La fricción es
el mecanismo que fuerza al modelo a admitir sus límites.

#!/usr/bin/env python3
import subprocess
import sys
import os
from datetime import datetime
from pathlib import Path

# CONFIGURACIÓN
WHISPER_BIN = os.path.expanduser("~/herramientas/whisper.cpp/build/bin/whisper-cli")
WHISPER_MODEL = os.path.expanduser("~/herramientas/whisper.cpp/models/ggml-small.bin")
DIR_BITACORA = Path(os.path.expanduser("~/proyecto/bitacora/audios"))
DURACION_MAX = 60
LIMITE_PALABRAS = 120

DIR_BITACORA.mkdir(parents=True, exist_ok=True)

def grabar_audio(segundos, nombre):
    archivo = DIR_BITACORA / f"{nombre}.wav"
    print(f"🔴 Grabando {segundos}s... Habla ahora.")
    subprocess.run(["arecord", "-f", "cd", "-t", "wav", "-d", str(segundos), str(archivo)], check=True)
    print(f"✅ Audio: {archivo}")
    return archivo

def transcribir(archivo_wav, idioma="es"):
    archivo_txt = archivo_wav.with_suffix(".txt")
    cmd = [WHISPER_BIN, "-m", WHISPER_MODEL, "-f", str(archivo_wav), "-l", idioma, "-ml", "80", "-otxt", "-of", str(archivo_wav.with_suffix(""))]
    print("🧠 Transcribiendo...")
    subprocess.run(cmd, check=True, capture_output=True)
    if not archivo_txt.exists():
        raise FileNotFoundError(f"No se generó: {archivo_txt}")
    texto = archivo_txt.read_text(encoding="utf-8").strip()
    return texto

def verificar_limite(texto, limite=LIMITE_PALABRAS):
    palabras = len(texto.split())
    print(f"📊 Palabras: {palabras} / {limite}")
    if palabras > limite:
        print("⚠️  EXCEDE el límite. Considera dividir en: Contexto + Petición.")
        return False
    print("✅ Dentro del límite.")
    return True

def guardar_en_bitacora(texto, archivo_audio):
    fecha = datetime.now().strftime("%Y-%m-%d_%H%M")
    entrada = DIR_BITACORA / f"entrada_{fecha}.md"
    contenido = f"""# Entrada de voz — {fecha}

**Audio:** `{archivo_audio.name}`
**Palabras:** {len(texto.split())}

---

{texto}
"""
    entrada.write_text(contenido, encoding="utf-8")
    print(f"📝 Bitácora: {entrada}")
    return entrada

def main():
    nombre = f"consulta_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
    try:
        audio = grabar_audio(DURACION_MAX, nombre)
        texto = transcribir(audio)
        print(f"\n{'─'*50}")
        print(texto)
        print(f"{'─'*50}\n")
        if verificar_limite(texto):
            guardar_en_bitacora(texto, audio)
        else:
            print("\n💡 Puedes editar el texto o dividirlo en dos entradas.")
    except Exception as e:
        print(f"❌ Error: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
PYEOF
