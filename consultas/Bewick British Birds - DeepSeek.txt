#!/usr/bin/env python3
"""
15_verificar_bibliografia.py — Verifica DOIs y recupera metadatos vía Crossref + OpenAlex.

Uso:
    python3 15_verificar_bibliografia.py --doi 10.1038/nrdp.2015.15
    python3 15_verificar_bibliografia.py --lista referencias.txt
    python3 15_verificar_bibliografia.py --csv referencias.csv

Salida:
    resultados/bibliografia_verificada.csv
"""

import argparse
import csv
import json
import time
from pathlib import Path

import requests


BASE = Path(__file__).resolve().parent.parent
OUT_DIR = BASE / "resultados"
OUT_CSV = OUT_DIR / "bibliografia_verificada.csv"

CROSSREF = "https://api.crossref.org/works"
OPENALEX = "https://api.openalex.org/works"
HEADERS = {"User-Agent": "Jacobea-Henoch-Study/0.1 (mailto:marco.juridico2026@gmail.com)"}


def consultar_crossref(doi: str) -> dict:
    url = f"{CROSSREF}/{doi}"
    try:
        r = requests.get(url, headers=HEADERS, timeout=30)
        if r.status_code == 200:
            data = r.json()["message"]
            return {
                "doi": doi,
                "titulo": data.get("title", [""])[0],
                "autores": "; ".join(
                    f"{a.get('family', '')}, {a.get('given', '')}"
                    for a in data.get("author", [])
                ),
                "anio": (data.get("published-print", {}).get("date-parts", [[None]])[0][0]
                         or data.get("published-online", {}).get("date-parts", [[None]])[0][0]
                         or data.get("created", {}).get("date-parts", [[None]])[0][0]),
                "revista": data.get("container-title", [""])[0],
                "volumen": data.get("volume", ""),
                "paginas": data.get("page", ""),
                "url": data.get("URL", ""),
                "tipo": data.get("type", ""),
            }
        else:
            return {"doi": doi, "error": f"Crossref status {r.status_code}"}
    except Exception as e:
        return {"doi": doi, "error": str(e)}


def consultar_openalex(doi: str) -> dict:
    url = f"{OPENALEX}/https://doi.org/{doi}"
    try:
        r = requests.get(url, headers=HEADERS, timeout=30)
        if r.status_code == 200:
            data = r.json()
            return {
                "openalex_id": data.get("id", ""),
                "abstract": data.get("abstract_inverted_index", {}),
                "citado_por": data.get("cited_by_count", 0),
                "es_retractado": data.get("is_retracted", False),
            }
        else:
            return {"error": f"OpenAlex status {r.status_code}"}
    except Exception as e:
        return {"error": str(e)}


def verificar_doi(doi: str) -> dict:
    print(f"Verificando: {doi}")
    cr = consultar_crossref(doi)
    oa = consultar_openalex(doi)
    return {**cr, **oa}


def reconstruir_abstract(inverted_index: dict) -> str:
    if not inverted_index:
        return ""
    posiciones = {}
    for palabra, pos in inverted_index.items():
        for p in pos:
            posiciones[p] = palabra
    return " ".join(posiciones[i] for i in sorted(posiciones))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--doi", type=str)
    ap.add_argument("--lista", type=Path)
    ap.add_argument("--csv", type=Path)
    args = ap.parse_args()

    OUT_DIR.mkdir(parents=True, exist_ok=True)

    dois = []
    if args.doi:
        dois = [args.doi]
    elif args.lista:
        dois = [l.strip() for l in args.lista.read_text(encoding="utf-8").splitlines()
                if l.strip() and not l.startswith("#")]
    elif args.csv:
        with args.csv.open(encoding="utf-8") as f:
            for row in csv.DictReader(f):
                d = row.get("doi", "").strip()
                if d:
                    dois.append(d)
    else:
        print("Uso: --doi <DOI> | --lista <archivo.txt> | --csv <archivo.csv>")
        return

    resultados = []
    for doi in dois:
        res = verificar_doi(doi)
        res["abstract"] = reconstruir_abstract(res.get("abstract", {}))
        resultados.append(res)
        time.sleep(1)

    campos = ["doi", "titulo", "autores", "anio", "revista", "volumen",
              "paginas", "tipo", "citado_por", "es_retractado", "abstract", "url", "error"]
    with OUT_CSV.open("w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=campos, extrasaction="ignore")
        w.writeheader()
        for r in resultados:
            w.writerow({k: r.get(k, "") for k in campos})

    print(f"\nEscrito: {OUT_CSV}")
    print(f"Total verificados: {len(resultados)}")
    for r in resultados:
        if "error" in r and r["error"]:
            print(f"  ❌ {r['doi']}: {r['error']}")
        else:
            print(f"  ✅ {r['doi']}: {r.get('titulo', '?')[:60]}")


if __name__ == "__main__":
    main()
